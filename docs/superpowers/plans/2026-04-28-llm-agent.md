# LLM Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Evoluir o chat e o upload do Financas para uma arquitetura de agente LLM com tool use, confirmação obrigatória antes de salvar qualquer transação (via chat ou foto), insights proativos e metas financeiras.

**Architecture:** O `ChatHandler` é estendido com um loop de tool use (máximo 5 iterações); o mecanismo de tags `<action:register_transaction>` é removido e substituído pela WriteTool `create_transaction`. O `UploadHandler` para de salvar transações diretamente — persiste um `PendingAction` com os dados do OCR e aguarda confirmação do usuário. Um `ToolRegistry` separa ferramentas de leitura de escrita. Um `InsightWorker` roda periodicamente e gera insights proativos.

**Tech Stack:** Go 1.25, Chi v5, GORM, PostgreSQL, Anthropic Messages API (tool use), React 18, TanStack React Query v5, Zustand.

```go
	return &PendingActionRepository{db: db}
}

func (r *PendingActionRepository) Create(pa *models.PendingAction) error {
	if err := r.db.Create(pa).Error; err != nil {
		return fmt.Errorf("pending action repository: create: %w", err)
	}
	return nil
}

// ClaimAndExecute atomically finds a valid pending action and marks it executed.
// Returns the action, or an error if not found, expired, or already executed.
func (r *PendingActionRepository) ClaimAndExecute(id, accountID uuid.UUID) (*models.PendingAction, error) {
	now := time.Now().UTC()
	var pa models.PendingAction
	result := r.db.Model(&pa).
		Where("id = ? AND account_id = ? AND executed_at IS NULL AND expires_at > ?", id, accountID, now).
		Update("executed_at", now)
	if result.Error != nil {
		return nil, fmt.Errorf("pending action repository: claim: %w", result.Error)
	}
	if result.RowsAffected == 0 {
		return nil, fmt.Errorf("pending action repository: claim: not found or expired")
	}
	if err := r.db.First(&pa, "id = ?", id).Error; err != nil {
		return nil, fmt.Errorf("pending action repository: fetch after claim: %w", err)
	}
	return &pa, nil
}
```

```go
	if unreadOnly {

	}
	err := q.Order("created_at desc").Find(&insights).Error

		return nil, fmt.Errorf("insight repository: list: %w", err)

func (r *InsightRepository) MarkRead(id, accountID uuid.UUID) error {
	result := r.db.Model(&models.AgentInsight{}).
		Where("id = ? AND account_id = ? AND read_at IS NULL", id, accountID).
		Update("read_at", time.Now().UTC())
	if result.Error != nil {
		return fmt.Errorf("insight repository: mark read: %w", result.Error)
	}
	return nil
}

// HasUnreadOfType checks if there is an unread insight of given type created in the last 24h.
func (r *InsightRepository) HasUnreadOfType(accountID uuid.UUID, insightType string) (bool, error) {
	var count int64
	since := time.Now().UTC().Add(-24 * time.Hour)
	err := r.db.Model(&models.AgentInsight{}).
		Where("account_id = ? AND type = ? AND read_at IS NULL AND created_at > ?", accountID, insightType, since).
		Count(&count).Error
	if err != nil {
		return false, fmt.Errorf("insight repository: has unread: %w", err)
	}
	return count > 0, nil
}
```

```go
func (p *AnthropicProvider) ChatWithTools(ctx context.Context, req ChatWithToolsRequest) (ChatWithToolsResponse, error) {
	msgs := make([]anthropicMessage, len(req.Messages))
	for i, m := range req.Messages {
		msgs[i] = anthropicMessage{
			Role:    m.Role,
			Content: []anthropicContent{{Type: "text", Text: m.Content}},
		}
	}

	tools := make([]anthropicToolDefinition, len(req.Tools))
	for i, t := range req.Tools {
		tools[i] = anthropicToolDefinition{
			Name:        t.Name,
			Description: t.Description,
			InputSchema: t.InputSchema,
		}
	}

	maxTokens := req.MaxTokens
	if maxTokens == 0 {
		maxTokens = 2048
	}

	body := anthropicRequestWithTools{
		Model:     req.Model,
		MaxTokens: maxTokens,
		System:    req.SystemPrompt,
		Messages:  msgs,
		Tools:     tools,
	}

	payload, err := json.Marshal(body)
	if err != nil {
		return ChatWithToolsResponse{}, fmt.Errorf("anthropic: marshal tools request: %w", err)
	}

	httpReq, err := http.NewRequestWithContext(ctx, http.MethodPost, anthropicAPIURL, bytes.NewReader(payload))
	if err != nil {
		return ChatWithToolsResponse{}, fmt.Errorf("anthropic: create tools request: %w", err)
	}
	httpReq.Header.Set("Content-Type", "application/json")
	httpReq.Header.Set("x-api-key", p.apiKey)
	httpReq.Header.Set("anthropic-version", anthropicVersion)

	resp, err := p.httpClient.Do(httpReq)
	if err != nil {
		return ChatWithToolsResponse{}, fmt.Errorf("anthropic: http tools request: %w", err)
	}
	defer resp.Body.Close()

	respBody, err := io.ReadAll(resp.Body)
	if err != nil {
		return ChatWithToolsResponse{}, fmt.Errorf("anthropic: read tools response: %w", err)
	}
	if resp.StatusCode != http.StatusOK {
		return ChatWithToolsResponse{}, fmt.Errorf("anthropic: api tools error %d: %s", resp.StatusCode, string(respBody))
	}

	var result anthropicResponseWithTools
	if err := json.Unmarshal(respBody, &result); err != nil {
		return ChatWithToolsResponse{}, fmt.Errorf("anthropic: unmarshal tools response: %w", err)
	}

	var textContent string
	var toolUses []ToolUseBlock
	for _, block := range result.Content {
		switch block.Type {
		case "text":
			textContent += block.Text
		case "tool_use":
			toolUses = append(toolUses, ToolUseBlock{
				ID:    block.ID,
				Name:  block.Name,
				Input: block.Input,
			})
		}
	}

	return ChatWithToolsResponse{
		Content:    textContent,
		StopReason: result.StopReason,
		ToolUse:    toolUses,
	}, nil
}
```

```go
package agent

import (
	"context"
	"encoding/json"
	"fmt"

	"github.com/google/uuid"
	"github.com/marianamohr/financas/internal/services/ai"
)

// ReadTool executes immediately without user confirmation.
type ReadTool interface {
	Name() string
	Schema() ai.ToolDefinition
	Execute(ctx context.Context, accountID uuid.UUID, input json.RawMessage) (string, error)
}

// WriteTool requires user confirmation before execution.
type WriteTool interface {
	Name() string
	Schema() ai.ToolDefinition
	Preview(input json.RawMessage) (string, error)
	Execute(ctx context.Context, accountID uuid.UUID, input json.RawMessage) (string, error)
}

// Registry holds all available tools.
type Registry struct {
	reads  map[string]ReadTool
	writes map[string]WriteTool
}

func NewRegistry(reads []ReadTool, writes []WriteTool) *Registry {
	r := &Registry{
		reads:  make(map[string]ReadTool, len(reads)),
		writes: make(map[string]WriteTool, len(writes)),
	}
	for _, t := range reads {
		r.reads[t.Name()] = t
	}
	for _, t := range writes {
		r.writes[t.Name()] = t
	}
	return r
}

func (r *Registry) Schemas() []ai.ToolDefinition {
	schemas := make([]ai.ToolDefinition, 0, len(r.reads)+len(r.writes))
	for _, t := range r.reads {
		schemas = append(schemas, t.Schema())
	}
	for _, t := range r.writes {
		schemas = append(schemas, t.Schema())
	}
	return schemas
}

func (r *Registry) IsWrite(name string) bool {
	_, ok := r.writes[name]
	return ok
}

func (r *Registry) ExecuteRead(ctx context.Context, accountID uuid.UUID, block ai.ToolUseBlock) (string, error) {
	t, ok := r.reads[block.Name]
	if !ok {
		return "", fmt.Errorf("registry: unknown read tool: %s", block.Name)
	}
	return t.Execute(ctx, accountID, block.Input)
}

func (r *Registry) PreviewWrite(name string, input json.RawMessage) (string, error) {
	t, ok := r.writes[name]
	if !ok {
		return "", fmt.Errorf("registry: unknown write tool: %s", name)
	}
	return t.Preview(input)
}

func (r *Registry) ExecuteWrite(ctx context.Context, accountID uuid.UUID, name string, input json.RawMessage) (string, error) {
	t, ok := r.writes[name]
	if !ok {
		return "", fmt.Errorf("registry: unknown write tool: %s", name)
	}
	return t.Execute(ctx, accountID, input)
}
```

```go
		Category string  `json:"category"`
		PeriodA  float64 `json:"period_a"`
		PeriodB  float64 `json:"period_b"`
		Delta    float64 `json:"delta"`
	}
	var deltas []delta
	all := map[string]bool{}
	for k := range a.Categories { all[k] = true }
	for k := range b.Categories { all[k] = true }
	for cat := range all {
		deltas = append(deltas, delta{
			Category: cat,
			PeriodA:  a.Categories[cat],
			PeriodB:  b.Categories[cat],
			Delta:    b.Categories[cat] - a.Categories[cat],
		})
	}
	out, _ := json.Marshal(map[string]any{"deltas": deltas, "total_a": a.Expenses, "total_b": b.Expenses})

func (t *detectAnomaliesTool) Name() string { return "detect_anomalies" }

func (t *detectAnomaliesTool) Schema() ai.ToolDefinition {
	return ai.ToolDefinition{
		Name:        "detect_anomalies",
		Description: "Detecta categorias com gastos fora do padrão histórico (último mês vs. média dos 3 anteriores).",
		InputSchema: map[string]any{"type": "object", "properties": map[string]any{}},
	}
}

func (t *detectAnomaliesTool) Execute(_ context.Context, accountID uuid.UUID, _ json.RawMessage) (string, error) {
	now := time.Now().UTC()
	current, err := t.repo.SummaryByMonth(accountID, now.Year(), int(now.Month()))
	if err != nil {
		return "", err
	}

	// média dos 3 meses anteriores por categoria
	categoryTotals := map[string][]float64{}
	for i := 1; i <= 3; i++ {
		prev := now.AddDate(0, -i, 0)
		s, err := t.repo.SummaryByMonth(accountID, prev.Year(), int(prev.Month()))
		if err != nil {
			continue
		}
		for cat, val := range s.Categories {
			categoryTotals[cat] = append(categoryTotals[cat], math.Abs(val))
		}
	}

	type anomaly struct {
		Category  string  `json:"category"`
		Current   float64 `json:"current"`
		AvgPast3  float64 `json:"avg_past_3"`
		PctChange float64 `json:"pct_change"`
	}
	var anomalies []anomaly
	for cat, curVal := range current.Categories {
		if curVal >= 0 {
			continue // ignora receitas
		}
		past := categoryTotals[cat]
		if len(past) == 0 {
			continue
		}
		var sum float64
		for _, v := range past { sum += v }
		avg := sum / float64(len(past))
		if avg == 0 {
			continue
		}
		cur := math.Abs(curVal)
		pct := (cur - avg) / avg * 100
		if pct > 30 { // anomalia: +30% acima da média
			anomalies = append(anomalies, anomaly{
				Category:  cat,
				Current:   math.Round(cur*100) / 100,
				AvgPast3:  math.Round(avg*100) / 100,
				PctChange: math.Round(pct*10) / 10,
			})
		}
	}
	out, _ := json.Marshal(anomalies)
	return string(out), nil
}
```

func (h *ChatHandler) Chat(w http.ResponseWriter, r *http.Request) {
	userID := auth.UserIDFromContext(r.Context())
	accountID := auth.AccountIDFromContext(r.Context())

	var req chatRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil || req.Message == "" {
		respondError(w, http.StatusBadRequest, "message is required")
		return
	}

	userMsg := &models.ChatMessage{
		UserID:    userID,
		AccountID: accountID,
		Role:      "user",
		Content:   req.Message,
	}
	if err := h.chatRepo.Save(userMsg); err != nil {
		log.Error().Err(err).Msg("chat: save user message")
		respondError(w, http.StatusInternalServerError, "internal error")
		return
	}

	user, err := h.userRepo.FindByID(userID)
	if err != nil {
		log.Error().Err(err).Msg("chat: fetch user")
		respondError(w, http.StatusInternalServerError, "internal error")
		return
	}

	systemPrompt, err := h.ragBuilder.Build(accountID, user.CategoryContext)
	if err != nil {
		log.Error().Err(err).Msg("chat: build rag context")
		respondError(w, http.StatusInternalServerError, "internal error")
		return
	}

	history, err := h.chatRepo.HistoryByAccount(accountID, chatHistoryLimit)
	if err != nil {
		log.Error().Err(err).Msg("chat: fetch history")
		respondError(w, http.StatusInternalServerError, "internal error")
		return
	}

	messages := make([]ai.Message, len(history))
	for i, m := range history {
		messages[i] = ai.Message{Role: m.Role, Content: m.Content}
	}

	finalReply, pendingAction, err := h.runAgentLoop(r.Context(), accountID, systemPrompt, messages)
	if err != nil {
		log.Error().Err(err).Msg("chat: agent loop")
		respondError(w, http.StatusInternalServerError, "internal error")
		return
	}

	assistantMsg := &models.ChatMessage{
		UserID:    userID,
		AccountID: accountID,
		Role:      "assistant",
		Content:   finalReply,
	}
	if err := h.chatRepo.Save(assistantMsg); err != nil {
		log.Error().Err(err).Msg("chat: save assistant message")
	}

	resp := chatResponse{Reply: finalReply}
	if pendingAction != nil {
		resp.PendingAction = pendingAction
	}
	respondJSON(w, http.StatusOK, resp)
}

func (h *ChatHandler) runAgentLoop(
	ctx context.Context,
	accountID uuid.UUID,
	systemPrompt string,
	messages []ai.Message,
) (string, *pendingActionResponse, error) {
	var finalContent string

	for i := 0; i < maxToolIterations; i++ {
		resp, err := h.aiProvider.ChatWithTools(ctx, ai.ChatWithToolsRequest{
			Model:        ai.SelectModel(ai.TaskFinancialChat),
			SystemPrompt: systemPrompt,
			Messages:     messages,
			Tools:        h.registry.Schemas(),
			MaxTokens:    1024,
		})
		if err != nil {
			return "", nil, fmt.Errorf("agent loop: call ai: %w", err)
		}

		finalContent = resp.Content

		if resp.StopReason == "end_turn" || len(resp.ToolUse) == 0 {
			break
		}

		// Adiciona a resposta do assistente ao histórico de mensagens
		assistantContent := resp.Content
		for _, block := range resp.ToolUse {
			assistantContent += fmt.Sprintf("\n[tool_use:%s]", block.Name)
		}
		messages = append(messages, ai.Message{Role: "assistant", Content: assistantContent})

		for _, block := range resp.ToolUse {
			if h.registry.IsWrite(block.Name) {
				preview, err := h.registry.PreviewWrite(block.Name, block.Input)
				if err != nil {
					log.Error().Err(err).Str("tool", block.Name).Msg("chat: preview write tool")
					preview = fmt.Sprintf("Executar %s", block.Name)
				}
				pa := &models.PendingAction{
					AccountID: accountID,
					ToolName:  block.Name,
					Payload:   []byte(block.Input),
					Preview:   preview,
					ExpiresAt: time.Now().UTC().Add(pendingActionTTL),
				}
				if err := h.pendingRepo.Create(pa); err != nil {
					return "", nil, fmt.Errorf("agent loop: save pending action: %w", err)
				}
				// Informa ao Claude que a ação está aguardando confirmação
				messages = append(messages, ai.Message{
					Role:    "user",
					Content: fmt.Sprintf("[tool_result:%s] Ação registrada e aguardando confirmação do usuário.", block.ID),
				})
				return finalContent, &pendingActionResponse{
					ID:       pa.ID.String(),
					ToolName: pa.ToolName,
					Preview:  pa.Preview,
				}, nil
			}

			result, err := h.registry.ExecuteRead(ctx, accountID, block)
			if err != nil {
				log.Error().Err(err).Str("tool", block.Name).Msg("chat: execute read tool")
				result = fmt.Sprintf("Erro ao executar %s", block.Name)
			}
			messages = append(messages, ai.Message{
				Role:    "user",
				Content: fmt.Sprintf("[tool_result:%s] %s", block.ID, result),
			})
		}
	}

	return finalContent, nil, nil
}

```go
// Start runs the worker in a goroutine. Cancel ctx to stop.
func (w *InsightWorker) Start(ctx context.Context) {
	go func() {
		w.run(ctx)
		ticker := time.NewTicker(w.interval)
		defer ticker.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				w.run(ctx)
			}
		}
	}()
}

func (w *InsightWorker) run(ctx context.Context) {
	accounts, err := w.accountRepo.ListActive()
	if err != nil {
		log.Error().Err(err).Msg("insight worker: list accounts")
		return
	}
	for _, accountID := range accounts {
		w.processAccount(ctx, accountID)
	}
}

func (w *InsightWorker) processAccount(ctx context.Context, accountID uuid.UUID) {
	now := time.Now().UTC()

	// Anomalias
	w.maybeGenerateInsight(ctx, accountID, "anomaly", func() (string, error) {
		return w.registry.reads["detect_anomalies"].Execute(ctx, accountID, json.RawMessage("{}"))
	}, func(data string) string {
		return fmt.Sprintf("Detectei gastos acima do padrão este mês: %s", data)
	})

	// Overspend vs mês anterior
	w.maybeGenerateInsight(ctx, accountID, "overspend", func() (string, error) {
		input, _ := json.Marshal(map[string]any{
			"year_a": now.Year(), "month_a": int(now.Month()) - 1,
			"year_b": now.Year(), "month_b": int(now.Month()),
		})
		return w.registry.reads["compare_periods"].Execute(ctx, accountID, input)
	}, func(data string) string {
		return fmt.Sprintf("Comparação com o mês anterior: %s", data)
	})

	// Projeção de saldo
	w.maybeGenerateInsight(ctx, accountID, "projection", func() (string, error) {
		summaryInput, _ := json.Marshal(map[string]any{"year": now.Year(), "month": int(now.Month())})
		return w.registry.reads["get_spending_summary"].Execute(ctx, accountID, summaryInput)
	}, func(data string) string {
		return fmt.Sprintf("Projeção de saldo para o fim do mês com base no ritmo atual: %s", data)
	})

func (w *InsightWorker) maybeGenerateInsight(

	)

	}

		AccountID: accountID,
		Type:      insightType,
		Body:      resp.Content,
	}); err != nil {

	}
}
```

```typescript
import { useCallback } from "react";
import { useChatStore, ChatMessage } from "@/store/chat";
import { api, PendingAction } from "@/services/api";

export function useChat() {
  const { messages, isLoading, addMessage, setLoading } = useChatStore();

  const sendMessage = useCallback(async (content: string) => {
    const userMsg: ChatMessage = {
      id: crypto.randomUUID(),
      role: "user",
      content,
      createdAt: new Date(),
    };
    addMessage(userMsg);
    setLoading(true);

    try {
      const data = await api.post<{ reply: string; pending_action?: PendingAction }>(
        "/chat",
        { message: content }
      );
      const assistantMsg: ChatMessage = {
        id: crypto.randomUUID(),
        role: "assistant",
        content: data.reply,
        createdAt: new Date(),
        pendingAction: data.pending_action ?? undefined,
      };
      addMessage(assistantMsg);
    } catch {
      addMessage({
        id: crypto.randomUUID(),
        role: "assistant",
        content: "Desculpe, ocorreu um erro. Tente novamente.",
        createdAt: new Date(),
      });
    } finally {
      setLoading(false);
    }
  }, [addMessage, setLoading]);

  const sendImage = useCallback(async (file: File, instruction?: string) => {
    const preview = URL.createObjectURL(file)
    addMessage({
      id: crypto.randomUUID(),
      role: "user",
      content: instruction ? `📸 Nota fiscal enviada\n_"${instruction}"_` : "📸 Nota fiscal enviada",
      imageUrl: preview,
      createdAt: new Date(),
    })
    setLoading(true)

    try {
      const data = await api.uploadImage(file, instruction)
      const pendingAction: PendingAction | undefined = data.pending_action ?? undefined
      addMessage({
        id: crypto.randomUUID(),
        role: "assistant",
        content: pendingAction
          ? `Extraí ${pendingAction.preview}. Confirma para salvar?`
          : "Não foi possível identificar transações com confiança suficiente.",
        createdAt: new Date(),
        pendingAction,
      })
    } catch (err) {
      const message = err instanceof Error ? err.message : "Erro desconhecido."
      addMessage({
        id: crypto.randomUUID(),
        role: "assistant",
        content: `Não consegui processar a imagem. ${message}`,
        createdAt: new Date(),
      })
    } finally {
      setLoading(false)
    }
  }, [addMessage, setLoading])

  return { messages, isLoading, sendMessage, sendImage };
}
```

- [ ] **Step 2: Criar `frontend/src/components/ConfirmationCard.tsx`**

```tsx
import { useState } from "react"
import { useMutation, useQueryClient } from "@tanstack/react-query"
import { api, PendingAction } from "@/services/api"
import { Button } from "@/components/ui/button"
import { Card, CardContent, CardFooter } from "@/components/ui/card"

interface ConfirmationCardProps {
  pendingAction: PendingAction
}

export function ConfirmationCard({ pendingAction }: ConfirmationCardProps) {
  const queryClient = useQueryClient()
  const [result, setResult] = useState<string | null>(null)

  const confirm = useMutation({
    mutationFn: () => api.confirmAction(pendingAction.id),
    onSuccess: (data) => {
      setResult(data.result)
      queryClient.invalidateQueries({ queryKey: ["transactions"] })
      queryClient.invalidateQueries({ queryKey: ["goals"] })
    },
  })

  const [cancelled, setCancelled] = useState(false)

  if (cancelled) {
    return (
      <Card className="mt-2 border-muted">
        <CardContent className="pt-4 text-sm text-muted-foreground">
          Ação cancelada.
        </CardContent>
      </Card>
    )
  }

  if (result) {
    return (
      <Card className="mt-2 border-green-500">
        <CardContent className="pt-4 text-sm text-green-700 dark:text-green-400">
          ✓ {result}
        </CardContent>
      </Card>
    )
  }

  return (
    <Card className="mt-2">
      <CardContent className="pt-4">
        <p className="text-sm font-medium">{pendingAction.preview}</p>
        <p className="text-xs text-muted-foreground mt-1">Confirma esta ação?</p>
      </CardContent>
      <CardFooter className="gap-2">
        <Button
          size="sm"
          onClick={() => confirm.mutate()}
          disabled={confirm.isPending}
        >
          {confirm.isPending ? "Confirmando..." : "Confirmar"}
        </Button>
        <Button
          size="sm"
          variant="outline"
          onClick={() => setCancelled(true)}
          disabled={confirm.isPending}
        >
          Cancelar
        </Button>
      </CardFooter>
    </Card>
  )
}
```
